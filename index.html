Inicializar PIN por defecto (1234) si no existe if (!localStorage.getItem('pin_v1')) { const hashDefault = CryptoJS.SHA256('1234').toString(); localStorage.setItem('pin_v1', hashDefault); } // 2. Verificar bloqueo inicial verificarBloqueo(); // 3. Event Listeners para teclado táctil document.querySelectorAll('.pin-keypad .key').forEach(btn => { btn.addEventListener('click', () => manejarTecla(btn.dataset.key)); }); // 4. Event Listener para teclado físico document.addEventListener('keydown', manejarTecladoFisico); } function verificarBloqueo() { const ahora = Date.now(); if (ahora < pinState.bloqueoHasta) { const segundosRestantes = Math.ceil((pinState.bloqueoHasta - ahora) / 1000); mostrarError(`Sistema bloqueado. Intente en ${segundosRestantes}s.`); deshabilitarTeclado(true); setTimeout(verificarBloqueo, 1000); } else { if (pinState.bloqueoHasta > 0) { localStorage.removeItem('bloqueo_hasta'); localStorage.setItem('intentos_pin', '0'); pinState.intentos = 0; pinState.bloqueoHasta = 0; } deshabilitarTeclado(false); mostrarError(''); } } function manejarTecla(key) { if (Date.now() < pinState.bloqueoHasta) return; if (key >= '0' && key <= '9') { if (pinState.ingresado.length < 4) { pinState.ingresado += key; actualizarDots(); if (pinState.ingresado.length === 4) { setTimeout(validarPin, 200); // Delay visual corto } } } else if (key === 'backspace') { pinState.ingresado = pinState.ingresado.slice(0, -1); actualizarDots(); mostrarError(''); } else if (key === 'confirm') { if (pinState.ingresado.length === 4) { validarPin(); } } } function manejarTecladoFisico(e) { if (!document.getElementById('vista-pin').classList.contains('activa')) return; if (e.key >= '0' && e.key <= '9') manejarTecla(e.key); else if (e.key === 'Backspace') manejarTecla('backspace'); else if (e.key === 'Enter') manejarTecla('confirm'); } function actualizarDots() { document.querySelectorAll('.pin-dot').forEach((dot, index) => { dot.classList.toggle('filled', index < pinState.ingresado.length); }); } function validarPin() { const hashIngresado = CryptoJS.SHA256(pinState.ingresado).toString(); const hashGuardado = localStorage.getItem('pin_v1'); if (hashIngresado === hashGuardado) { // Éxito pinState.intentos = 0; localStorage.setItem('intentos_pin', '0'); localStorage.setItem('ultimo_login', Date.now().toString()); pinState.ingresado = ''; actualizarDots(); mostrarError(''); mostrarVista('licencia-requerida'); } else { // Fallo pinState.intentos++; localStorage.setItem('intentos_pin', pinState.intentos.toString()); pinState.ingresado = ''; actualizarDots(); if (pinState.intentos >= 5) { pinState.bloqueoHasta = Date.now() + (5 * 60 * 1000); // 5 minutos localStorage.setItem('bloqueo_hasta', pinState.bloqueoHasta.toString()); verificarBloqueo(); } else { mostrarError(`PIN incorrecto. Intentos: ${pinState.intentos}/5`); } } } function mostrarError(msg) { const el = document.getElementById('pin-error'); if (el) el.textContent = msg; } function deshabilitarTeclado(bloqueado) { document.querySelectorAll('.pin-keypad .key').forEach(btn => { btn.disabled = bloqueado; }); } // Expuesta globalmente para otros paneles (ej.
